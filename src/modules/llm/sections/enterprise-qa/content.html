SQL 
# MySQL表结构

# 用户表
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    department VARCHAR(50),
    role ENUM('admin', 'manager', 'employee') DEFAULT 'employee',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

# 文档表
CREATE TABLE documents (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    content TEXT,
    category VARCHAR(50),
    department VARCHAR(50),
    access_level ENUM('public', 'internal', 'confidential') DEFAULT 'internal',
    created_by INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id)
);

# 对话历史表
CREATE TABLE chat_history (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    session_id VARCHAR(100),
    role ENUM('user', 'assistant'),
    content TEXT,
    sources JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

from pymilvus import MilvusClient, CollectionSchema, FieldSchema, DataType

client = MilvusClient("./enterprise_rag.db")

fields = [
    FieldSchema("id", DataType.VARCHAR, is_primary=True, max_length=64),
    FieldSchema("doc_id", DataType.INT64),
    FieldSchema("embedding", DataType.FLOAT_VECTOR, dim=768),
    FieldSchema("chunk_text", DataType.VARCHAR, max_length=65535),
    FieldSchema("doc_type", DataType.VARCHAR, max_length=32),
]
schema = CollectionSchema(fields, description="企业文档向量库")

client.create_collection(
    collection_name="enterprise_docs",
    schema=schema
)

# 创建向量索引（必须在 insert 之前完成）
index_params = client.prepare_index_params()
index_params.add_index(
    field_name="embedding",
    index_type="HNSW",
    metric_type="COSINE",
    params={"M": 16, "efConstruction": 256}
)
client.create_index("enterprise_docs", index_params)

文档分块策略

高质量的文档分块是 RAG 系统效果的关键。推荐使用递归字符分割（Recursive Character Splitting）：

Python
from langchain_text_splitters import RecursiveCharacterTextSplitter

splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,       # 每块约500字符（中文约250字）
    chunk_overlap=50,      # 块间重叠50字符，避免语义截断
    separators=["\n\n", "\n", "。", "；", " ", ""],  # 中文友好的分隔符优先级
)

chunks = splitter.split_text(document_content)

分块参数推荐

场景chunk_sizechunk_overlap说明
通用文档50050平衡检索精度与上下文完整性
FAQ/知识库30030短文本匹配更精确
技术手册800100保留更多上下文关联
法律/合同1000150段落完整性优先

注意事项：chunk_size 不是越大越好。过大的块会稀释语义信号，降低检索精度；过小的块会丢失上下文，影响生成质量。建议在实际数据上做 A/B 测试。

混合检索实现

Python
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

class EnterpriseRAG:
    """企业级RAG系统"""

    def __init__(self):
        self.embed_model = SentenceTransformer("BAAI/bge-base-zh-v1.5")
        self.milvus = MilvusClient("./enterprise_rag.db")
        self.mysql = mysql.connector.connect(...)
        self.reranker = CrossEncoder("BAAI/bge-reranker-base")

        # LLM 初始化（使用 LangChain LCEL 模式）
        self.llm = ChatOpenAI(model="gpt-4o", temperature=0)
        self.prompt = ChatPromptTemplate.from_template(
            """基于以下文档回答问题。请引用来源。

文档：
{context}

问题：{question}

回答（请标注引用来源）："""
        )
        self.chain = self.prompt | self.llm | StrOutputParser()

    def get_access_filter(self, user_role):
        """根据用户角色返回权限过滤条件"""
        # TODO: 实现权限过滤逻辑
        if user_role == "admin":
            return "1=1"  # 管理员可访问全部
        elif user_role == "manager":
            return "access_level IN ('public', 'internal')"
        return "access_level = 'public'"

    def mysql_search(self, query, access_filter, limit=10):
        """MySQL 全文检索（MATCH...AGAINST）"""
        cursor = self.mysql.cursor(dictionary=True)
        sql = f"""
            SELECT id, title, content, category,
                   MATCH(title, content) AGAINST('{query}' IN NATURAL LANGUAGE MODE) AS score
            FROM documents
            WHERE {access_filter}
            ORDER BY score DESC
            LIMIT {limit}
        """
        cursor.execute(sql)
        return cursor.fetchall()

    def vector_search(self, query, access_filter, limit=10):
        """Milvus 向量相似度检索"""
        query_vec = self.embed_model.encode([query])[0].tolist()
        results = self.milvus.search(
            collection_name="enterprise_docs",
            data=[query_vec],
            limit=limit,
            output_fields=["chunk_text", "doc_id", "doc_type"],
        )
        # TODO: 结合 access_filter 做后过滤（或在 Milvus filter 表达式中实现）
        return [
            {"id": hit["id"], "content": hit["entity"]["chunk_text"],
             "doc_id": hit["entity"]["doc_id"], "score": hit["distance"]}
            for hit in results[0]
        ]

    def rrf_fusion(self, results_a, results_b, k=60):
        """Reciprocal Rank Fusion 融合两路检索结果"""
        scores = {}
        for rank, doc in enumerate(results_a):
            doc_id = doc.get("id") or doc.get("doc_id")
            scores[doc_id] = scores.get(doc_id, {"doc": doc, "score": 0})
            scores[doc_id]["score"] += 1.0 / (k + rank + 1)
        for rank, doc in enumerate(results_b):
            doc_id = doc.get("id") or doc.get("doc_id")
            scores[doc_id] = scores.get(doc_id, {"doc": doc, "score": 0})
            scores[doc_id]["score"] += 1.0 / (k + rank + 1)
        ranked = sorted(scores.values(), key=lambda x: x["score"], reverse=True)
        return [item["doc"] for item in ranked]

    def rerank(self, query, candidates, top_n=5):
        """Cross-Encoder 重排序"""
        pairs = [(query, doc.get("content", "")) for doc in candidates]
        scores = self.reranker.predict(pairs)
        for doc, score in zip(candidates, scores):
            doc["rerank_score"] = float(score)
        ranked = sorted(candidates, key=lambda x: x["rerank_score"], reverse=True)
        return ranked[:top_n]

    def retrieve(self, query, user_role, top_k=5):
        """混合检索 + 权限过滤"""

        # 1. 权限过滤SQL
        access_filter = self.get_access_filter(user_role)

        # 2. MySQL检索
        mysql_results = self.mysql_search(query, access_filter)

        # 3. 向量检索
        vector_results = self.vector_search(query, access_filter)

        # 4. RRF融合
        fused = self.rrf_fusion(mysql_results, vector_results)

        # 5. 重排序
        reranked = self.rerank(query, fused)

        return reranked[:top_k]

    def answer(self, query, user_role, session_id):
        """完整RAG问答"""

        # 检索相关文档
        docs = self.retrieve(query, user_role)

        # 构建上下文
        context = "\n\n".join([d["content"] for d in docs])

        # LCEL 链式调用生成回答
        response = self.chain.invoke({
            "context": context,
            "question": query
        })

        # 保存对话历史
        self.save_history(session_id, query, response, docs)

        return {
            "answer": response,
            "sources": docs,
            "session_id": session_id
        }
