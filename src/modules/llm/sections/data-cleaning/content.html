文本清洗

类比：文本清洗就像是做饭前的洗菜——把烂叶子（乱码）、泥土（HTML标签）、虫子（特殊字符）都去掉，只留下干净的食材。

常见清洗操作

Python 
import re
from bs4 import BeautifulSoup

def clean_text(text: str) -> str:
    """文本清洗主函数"""

    # 1. 去除HTML标签
    text = BeautifulSoup(text, "html.parser").get_text()

    # 2. 去除URL链接
    text = re.sub(r'https?://\S+|www\.\S+', '', text)

    # 3. 去除邮箱地址
    text = re.sub(r'\S+@\S+\.\S+', '', text)

    # 4. 去除多余空白字符
    text = re.sub(r'\s+', ' ', text)

    # 5. 去除特殊字符（保留中文、英文、数字、基本标点）
    text = re.sub(r'[^\w\s一-鿿.,!?;:\'"-]', '', text)

    # 6. 去除前后空白
    text = text.strip()

    return text

质量过滤

Python 
def quality_filter(text: str, min_length: int = 50, max_length: int = 10000) -> bool:
    """质量过滤：检查文本是否满足训练要求"""

    # 1. 长度过滤
    if len(text) < min_length or len(text) > max_length:
        return False

    # 2. 重复率检查（简单的n-gram重复）
    words = text.split()
    if len(words) > 100:
        unique_ratio = len(set(words)) / len(words)
        if unique_ratio < 0.3:  # 重复率超过70%
            return False

    # 3. 语言检测（推荐使用fastText的lid.176模型，支持176种语言）
    chinese_chars = sum(1 for c in text if '一' <= c <= '鿿')
    if len(text) > 0 and chinese_chars / len(text) > 0.1:
        return True  # 包含足够中文（注：生产环境推荐用fastText lid.176替代）

    # 4. 检查是否是乱码（连续特殊字符过多）
    special_count = sum(1 for c in text if not c.isalnum() and not c.isspace())
    if special_count / len(text) > 0.3:
        return False  # 可能是乱码

    return True

语言检测推荐：生产环境建议使用 fastText lid.176 模型进行语言识别（支持176种语言），准确率远高于简单的字符比例判断。FineWeb、RedPajama、Dolma等主流数据集均采用此方案。

困惑度（Perplexity）质量过滤

除了规则过滤外，还可以使用语言模型困惑度来评估文本质量——低困惑度的文本通常更通顺、更有价值。

Python 
# 使用KenLM进行困惑度过滤（CCNet/FineWeb方案）
# 安装: pip install kenlm
import kenlm

# 加载预训练的KenLM n-gram模型
model = kenlm.Model("zh.arpa.klm")

def perplexity_filter(text, max_ppl=1000):
    """困惑度过滤：文本越通顺，困惑度越低"""
    log_score = model.score(text, bos=True, eos=True)
    num_words = len(text.split())
    ppl = 10 ** (-log_score / num_words)
    return ppl < max_ppl  # 困惑度过高说明文本质量差

PII（个人身份信息）去除

Python 
import re

def remove_pii(text: str) -> str:
    """去除个人身份信息（PII）"""

    # 去除电话号码
    text = re.sub(r'1[3-9]\d{9}', '[电话]', text)  # 中国手机号
    text = re.sub(r'\d{3}-\d{4}-\d{4}', '[电话]', text)  # 座机

    # 去除身份证号
    text = re.sub(r'\d{17}[\dX]', '[身份证]', text)

    # 去除邮箱
    text = re.sub(r'\S+@\S+\.\S+', '[邮箱]', text)

    # 去除银行卡号
    text = re.sub(r'\d{16,19}', '[卡号]', text)

    return text

PII去除推荐：仅使用正则表达式容易遗漏边界情况。生产环境建议使用微软开源的 Presidio 库（pip install presidio-analyzer presidio-anonymizer），它结合了正则、NLP实体识别和上下文分析，能更准确地识别姓名、地址、信用卡号等PII信息。

文本分词（Tokenization）

类比：分词就像是把一篇文章拆成一个个单词或字符。但LLM使用的分词更智能——它不是按空格分词，而是学习最有用的子词单元（如"un"、"##ing"等）。

BPE（Byte-Pair Encoding）算法

BPE的核心思想：从字符开始，反复合并最频繁出现的相邻字符对。

BPE训练过程

Python 
from tokenizers import Tokenizer
from tokenizers.models import BPE
from tokenizers.trainers import BpeTrainer
from tokenizers.pre_tokenizers import Whitespace

# 1. 初始化BPE分词器
tokenizer = Tokenizer(BPE(unk_token="[UNK]"))
tokenizer.pre_tokenizer = Whitespace()

# 2. 配置训练器
trainer = BpeTrainer(
    vocab_size=30000,        # 词表大小
    min_frequency=2,         # 最小出现次数
    special_tokens=["[UNK]", "[CLS]", "[SEP]", "[PAD]", "[MASK]"]
)

# 3. 训练BPE
files = ["data/corpus.txt"]  # 训练语料文件列表
tokenizer.train(files, trainer)

# 4. 保存分词器
tokenizer.save("bpe_tokenizer.json")

# 5. 使用分词器
output = tokenizer.encode("Hello, how are you?")
print(output.tokens)  # ['Hello', ',', 'how', 'are', 'you', '?']
print(output.ids)     # [152, 15, 567, 389, 456, 35]

WordPiece算法

WordPiece与BPE类似，但合并标准不同：BPE选择频率最高的字符对，WordPiece选择使语言模型似然最大的字符对。

tokenizer = AutoTokenizer.from_pretrained("bert-base-chinese")

    max_length=128,
    truncation=True

    ["文本1", "这是一个很长的文本"],
    padding="longest",  # 填充到batch中最长的长度
    truncation=True
)

转换代码

import json

# 原始文本 → JSONL格式
def convert_to_jsonl(input_file, output_file):
    with open(input_file, 'r', encoding='utf-8') as f:
        lines = f.readlines()

    with open(output_file, 'w', encoding='utf-8') as f:
        for line in lines:
            line = line.strip()
            if line:
                json_line = {"text": line}
                f.write(json.dumps(json_line, ensure_ascii=False) + '\n')

统计分析

Python 
import json
from collections import Counter
import matplotlib.pyplot as plt

def analyze_dataset(jsonl_file):
    """分析数据集质量"""
    lengths = []
    sources = []

    with open(jsonl_file, 'r', encoding='utf-8') as f:
        for line in f:
            data = json.loads(line)
            lengths.append(len(data["text"]))
            sources.append(data.get("source", "unknown"))

    # 统计信息
    print(f"总样本数: {len(lengths)}")
    print(f"平均长度: {sum(lengths)/len(lengths):.1f}")
    print(f"长度中位数: {sorted(lengths)[len(lengths)//2]}")
    print(f"来源分布: {Counter(sources)}")

    # 可视化
    plt.figure(figsize=(10, 4))
    plt.hist(lengths, bins=50)
    plt.xlabel('文本长度')
    plt.ylabel('样本数')
    plt.title('文本长度分布')
    plt.savefig('length_distribution.png')
    plt.show()

重复检测

Python 
import hashlib
from collections import defaultdict

def detect_duplicates(jsonl_file):
    """检测重复文本"""
    hash_counts = defaultdict(int)
    duplicates = []

    with open(jsonl_file, 'r', encoding='utf-8') as f:
        for line_num, line in enumerate(f, 1):
            data = json.loads(line)
            text = data["text"]
            text_hash = hashlib.md5(text.encode()).hexdigest()

            hash_counts[text_hash] += 1
            if hash_counts[text_hash] > 1:
                duplicates.append((line_num, text[:50]))

    print(f"发现 {len(duplicates)} 个重复")
    return duplicates

MinHash + LSH 近似去重

上面的哈希方法只能检测完全相同的文本。实际数据中更多是近似重复（如略有修改的新闻转载、论坛引用）。主流数据集（FineWeb、RedPajama、Dolma）均采用 MinHash + LSH（局部敏感哈希） 进行大规模近似去重。

Python 
# MinHash + LSH 近似去重（推荐方案）
# 安装: pip install datasketch
from datasketch import MinHash, MinHashLSH

def create_minhash(text, num_perm=128):
    """将文本转为MinHash签名"""
    m = MinHash(num_perm=num_perm)
    for word in text.split():
        m.update(word.encode('utf-8'))
    return m

# 建立LSH索引（阈值0.8表示Jaccard相似度>0.8视为重复）
lsh = MinHashLSH(threshold=0.8, num_perm=128)

# 插入数据并查询近似重复
for idx, text in enumerate(texts):
    mh = create_minhash(text)
    result = lsh.query(mh)  # 查询是否有相似文档
    if not result:
        lsh.insert(f"doc_{idx}", mh)
    else:
        print(f"文档 {idx} 与 {result} 近似重复")

主流去重工具推荐：

  • text-dedup：集成多种去重算法（MinHash、SimHash、Suffix Array等），开箱即用
  • datatrove（HuggingFace）：大规模数据处理pipeline，包含MinHash去重模块
  • NVIDIA NeMo Curator：GPU加速的去重和质量过滤，适合超大规模数据
  • Dolma Toolkit（AI2）：Allen AI开源，被Dolma/OLMo数据集使用
