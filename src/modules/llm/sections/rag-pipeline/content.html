from langchain_community.document_loaders import DirectoryLoader, TextLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from sentence_transformers import SentenceTransformer
from pymilvus import MilvusClient
from pymilvus import FieldSchema, CollectionSchema, DataType, Collection
import mysql.connector
import uuid

class RAGPipeline:
    """完整RAG Pipeline"""

    def __init__(self):
        # 初始化组件
        self.embed_model = SentenceTransformer("BAAI/bge-base-zh-v1.5")
        self.milvus = MilvusClient("./rag_milvus.db")
        self.mysql = mysql.connector.connect(
            host="localhost", user="root",
            password="password", database="rag_db"
        )
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=500, chunk_overlap=50
        )

        # 创建Milvus集合（如不存在）
        if not self.milvus.has_collection("documents"):
            fields = [
                FieldSchema(name="id", dtype=DataType.VARCHAR,
                            is_primary=True, max_length=64),
                FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR,
                            dim=768),
                FieldSchema(name="title", dtype=DataType.VARCHAR,
                            max_length=512),
            ]
            schema = CollectionSchema(fields, description="RAG文档集合")
            self.milvus.create_collection(
                collection_name="documents",
                schema=schema
            )
            # 创建向量索引以加速检索
            index_params = self.milvus.prepare_index_params()
            index_params.add_index(
                field_name="embedding",
                index_type="IVF_FLAT",
                metric_type="COSINE",
                params={"nlist": 128}
            )
            self.milvus.create_index("documents", index_params)

    def ingest_documents(self, doc_dir: str):
        """数据导入Pipeline"""
        # 1. 加载文档
        loader = DirectoryLoader(doc_dir, glob="**/*.txt", loader_cls=TextLoader)
        documents = loader.load()

        # 2. 文本分割
        chunks = self.text_splitter.split_documents(documents)

        # 3. 写入MySQL（原始文档）
        cursor = self.mysql.cursor()
        for chunk in chunks:
            cursor.execute(
                "INSERT INTO documents (title, content, source) VALUES (%s, %s, %s)",
                (chunk.metadata["source"], chunk.page_content, chunk.metadata["source"])
            )
        self.mysql.commit()

        # 4. 生成向量并写入Milvus
        texts = [c.page_content for c in chunks]
        embeddings = self.embed_model.encode(texts).tolist()

        data = [
            {
                "id": str(uuid.uuid4()),  # 使用UUID避免重复导入时ID冲突
                "embedding": emb,
                "title": chunks[i].metadata["source"]
            }
            for i, emb in enumerate(embeddings)
        ]
        self.milvus.insert(collection_name="documents", data=data)

        print(f"✅ 导入完成：{len(chunks)} 个文档块")

import time
import logging

logger = logging.getLogger(__name__)

def query(self, question: str, top_k: int = 5) -> dict:
    """RAG查询（含错误处理与重试）"""

    # 1. 混合检索（含异常处理）
    try:
        results = self.hybrid_retrieve(question, top_k=top_k)
    except Exception as e:
        logger.error(f"检索失败：{e}")
        return {"answer": "抱歉，检索服务暂时不可用。", "sources": [], "context": ""}

    if not results:
        return {"answer": "未找到相关信息。", "sources": [], "context": ""}

    # 2. 构建Prompt
    context = "\n\n".join([r["content"] for r in results])
    prompt = f"""基于以下上下文回答问题。如果上下文中没有相关信息，请说"我不知道"。

上下文：
{context}

问题：{question}

回答："""

    # 3. LLM生成（带指数退避重试）
    max_retries = 3
    for attempt in range(max_retries):
        try:
            response = self.llm.generate(prompt, timeout=30)
            break
        except TimeoutError:
            logger.warning(f"LLM调用超时，第{attempt+1}次重试")
            time.sleep(2 ** attempt)  # 指数退避: 1s, 2s, 4s
        except Exception as e:
            logger.error(f"LLM调用失败：{e}")
            response = "抱歉，生成服务暂时不可用。"
            break

    return {
        "answer": response,
        "sources": results,
        "context": context
    }

def query_stream(self, question: str, top_k: int = 5):
    """流式RAG查询——逐token返回，提升用户体验"""
    results = self.hybrid_retrieve(question, top_k=top_k)
    if not results:
        yield "未找到相关信息。"
        return

    context = "\n\n".join([r["content"] for r in results])
    prompt = f"""基于以下上下文回答问题。

上下文：
{context}

问题：{question}

回答："""

    # 使用LLM流式接口逐token输出
    for token in self.llm.stream(prompt):
        yield token

# 使用示例
pipeline = RAGPipeline()
pipeline.ingest_documents("./docs")

# 普通查询
result = pipeline.query("什么是机器学习？")
print(result["answer"])

# 流式查询（适合Web前端实时展示）
for token in pipeline.query_stream("什么是机器学习？"):
    print(token, end="", flush=True)

RAG评估（RAGAS）

构建RAG系统后，如何量化评估其质量？RAGAS（Retrieval Augmented Generation Assessment）是最常用的RAG评估框架，提供以下核心指标：

指标含义评估对象
Faithfulness（忠实度）生成的答案是否基于检索到的上下文？有无幻觉？生成质量
Answer Relevancy（答案相关性）答案是否真正回答了用户的问题？生成质量
Context Precision（上下文精确率）相关文档是否被排在检索结果的前列？检索质量
Context Recall（上下文召回率）所有相关文档是否都被检索到了？检索质量

# 使用RAGAS评估RAG系统
# pip install ragas

from ragas import evaluate
from ragas.metrics import (
    faithfulness,
    answer_relevancy,
    context_precision,
    context_recall
)
from datasets import Dataset

# 准备评估数据集
eval_data = {
    "question": ["什么是机器学习？", "深度学习和机器学习的关系？"],
    "answer": ["机器学习是AI的分支...", "深度学习是机器学习的子集..."],
    "contexts": [
        ["机器学习是人工智能的一个分支", "它使计算机能够从数据中学习"],
        ["深度学习是机器学习的一个子领域", "使用多层神经网络"],
    ],
    "ground_truth": ["机器学习是AI的分支...", "深度学习是机器学习的子集..."]
}
dataset = Dataset.from_dict(eval_data)

# 运行评估
result = evaluate(
    dataset,
    metrics=[faithfulness, answer_relevancy, context_precision, context_recall],
)

print(result)  # 输出各指标分数（0-1）
result.to_pandas()  # 转为DataFrame查看详细结果

评估最佳实践：准备至少20-50个有标准答案（ground_truth）的测试用例，覆盖常见问题、边界情况和长尾问题。持续评估、持续迭代是提升RAG系统质量的关键。
