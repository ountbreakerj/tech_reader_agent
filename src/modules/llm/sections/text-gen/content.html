数据准备

Python 
from datasets import load_dataset

# 加载数据集
dataset = load_dataset("json", data_files={"train": "data/train.jsonl"})

# 数据格式示例
# {"text": "机器学习是人工智能的一个分支，它使计算机..."}

# 分词
from transformers import GPT2Tokenizer

tokenizer = GPT2Tokenizer.from_pretrained("uer/gpt2-chinese-cluecorpussmall")
# ⚠️ 不要用 eos_token 作为 pad_token，否则模型会把填充当作真实内容学习
tokenizer.add_special_tokens({"pad_token": "[PAD]"})

def tokenize_function(examples):
    return tokenizer(
        examples["text"],
        truncation=True,
        max_length=512,
        padding="max_length"
    )

tokenized_dataset = dataset.map(tokenize_function, batched=True)

模型微调

Python
from transformers import GPT2LMHeadModel, TrainingArguments, Trainer

# 加载模型
model = GPT2LMHeadModel.from_pretrained("uer/gpt2-chinese-cluecorpussmall")
# 调整词表大小以匹配新增的 [PAD] token
model.resize_token_embeddings(len(tokenizer))

# 训练参数
training_args = TrainingArguments(
    output_dir="./gpt2-chinese",
    num_train_epochs=3,
    per_device_train_batch_size=8,
    learning_rate=5e-5,
    warmup_steps=100,
    logging_steps=50,
    save_strategy="epoch",
    eval_strategy="epoch",
    load_best_model_at_end=True,
    metric_for_best_model="eval_loss",
)

# 创建Trainer
trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_dataset["train"],
    eval_dataset=tokenized_dataset["validation"],
)

# 开始训练
trainer.train()

生成效果测试

Python
from transformers import pipeline
import torch

# 设备管理
device = "cuda" if torch.cuda.is_available() else "cpu"

# load_best_model_at_end=True 时，最佳模型保存在 output_dir 根目录
generator = pipeline(
    "text-generation",
    model="./gpt2-chinese",
    tokenizer=tokenizer,
    device=device
)

# 测试生成
prompts = [
    "机器学习的主要方法包括",
    "深度学习在自然语言处理中的应用",
    "人工智能的未来发展趋势",
]

for prompt in prompts:
    result = generator(
        prompt,
        max_new_tokens=200,
        num_return_sequences=1,
        temperature=0.7,
        do_sample=True,
        top_k=50,
        top_p=0.95
    )
    print(f"提示：{prompt}")
    print(f"生成：{result[0]['generated_text']}")
    print("---")

生成参数说明

参数说明推荐值
temperature控制随机性，越高越随机0.7
top_k只从概率最高的K个词中采样50
top_pnucleus sampling，累计概率阈值0.95
max_new_tokens最多生成的新token数（替代已废弃的 max_length）200

API部署

Python
from fastapi import FastAPI
from pydantic import BaseModel
from transformers import pipeline
import torch

app = FastAPI()

# GPU/设备管理
device = "cuda" if torch.cuda.is_available() else "cpu"
generator = pipeline("text-generation", model="./gpt2-chinese", device=device)

class GenerateRequest(BaseModel):
    prompt: str
    max_new_tokens: int = 200
    temperature: float = 0.7

@app.post("/generate")
async def generate(request: GenerateRequest):
    result = generator(
        request.prompt,
        max_new_tokens=request.max_new_tokens,
        temperature=request.temperature,
        do_sample=True
    )
    return {"text": result[0]["generated_text"]}

# 启动：uvicorn api:app --reload

现代替代方案

2026年推荐：以下方案在效率和效果上都优于原始 GPT-2 微调。

1. PEFT/LoRA — 参数高效微调

对于 GPT-2 这类自回归模型，LoRA 可以将训练参数量减少 90% 以上：

Python
from peft import LoraConfig, get_peft_model, TaskType

config = LoraConfig(
    task_type=TaskType.CAUSAL_LM,
    r=16,
    lora_alpha=32,
    lora_dropout=0.05,
    target_modules=["c_attn", "c_proj"],
)
model = get_peft_model(model, config)
