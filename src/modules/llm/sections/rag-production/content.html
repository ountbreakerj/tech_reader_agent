async def rag_query(request: QueryRequest):
    """RAG 查询主流程"""
    # 1. 检查缓存
    cached = await app.state.cache.get(request.question)
    if cached:
        return CachedResponse(answer=cached, from_cache=True)

    # 2. 查询预处理
    rewritten = await rewrite_query(request.question)

    # 3. 检索
    contexts = await app.state.vector_store.search(
        query=rewritten, top_k=request.top_k
    )

    # 4. 生成
    answer = await app.state.llm_client.generate(
        question=request.question, contexts=contexts
    )

    # 5. 写入缓存
    await app.state.cache.set(request.question, answer)

    return RAGResponse(answer=answer, contexts=contexts, from_cache=False)

2. 缓存策略

缓存是降低RAG系统延迟和成本最有效的手段。一个好的缓存策略可以将LLM调用量减少30%-70%。

语义缓存

语义缓存通过embedding相似度匹配，即使用户换了一种问法也能命中缓存。例如"Python怎么读文件"和"如何用Python读取文件"语义相同。

import hashlib
import numpy as np
from sentence_transformers import SentenceTransformer

class SemanticCache:
    """基于embedding相似度的语义缓存"""

    def __init__(self, similarity_threshold: float = 0.95):
        self.model = SentenceTransformer("all-MiniLM-L6-v2")
        self.entries: list[dict] = []  # {"query", "embedding", "answer", "hit_count"}
        self.threshold = similarity_threshold

    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

    def get(self, query: str) -> str | None:
        """查询缓存，返回命中答案或None"""
        if not self.entries:
            return None
        query_emb = self.model.encode(query)
        best_score = 0.0
        best_entry = None
        for entry in self.entries:
            score = self._cosine_similarity(query_emb, entry["embedding"])
            if score > best_score:
                best_score = score
                best_entry = entry
        if best_score >= self.threshold:
            best_entry["hit_count"] += 1
            return best_entry["answer"]
        return None

    def set(self, query: str, answer: str):
        """写入缓存"""
        embedding = self.model.encode(query)
        self.entries.append({
            "query": query,
            "embedding": embedding,
            "answer": answer,
            "hit_count": 0,
        })

# 使用示例
cache = SemanticCache(similarity_threshold=0.92)

# 第一次查询 - 缓存未命中
result = cache.get("什么是向量数据库？")  # None
cache.set("什么是向量数据库？", "向量数据库是专门存储...")

# 语义相似查询 - 缓存命中
result = cache.get("向量数据库是什么？")  # "向量数据库是专门存储..."

精确缓存

精确缓存使用query的hash作为key，速度极快但无法处理同义问法。

import hashlib
import json
import redis

class ExactCache:
    """基于Redis的精确缓存"""

    def __init__(self, redis_url: str = "redis://localhost:6379", ttl: int = 3600):
        self.redis = redis.from_url(redis_url)
        self.ttl = ttl  # 过期时间（秒）

    def _make_key(self, query: str, model: str = "default") -> str:
        """生成缓存键"""
        content = f"{model}:{query.strip().lower()}"
        return f"rag_cache:{hashlib.sha256(content.encode()).hexdigest()}"

    def get(self, query: str, model: str = "default") -> str | None:
        key = self._make_key(query, model)
        result = self.redis.get(key)
        return result.decode("utf-8") if result else None

    def set(self, query: str, answer: str, model: str = "default"):
        key = self._make_key(query, model)
        self.redis.setex(key, self.ttl, answer)

    def invalidate(self, query: str, model: str = "default"):
        """手动失效缓存"""
        key = self._make_key(query, model)
        self.redis.delete(key)

3. 流式输出

对于RAG系统，用户可能需要等待检索+生成的时间。流式输出可以让用户看到逐步生成的文字，大幅改善感知延迟。

from fastapi import FastAPI
from fastapi.responses import StreamingResponse

import json

app = FastAPI()

async def stream_rag_response(question: str):
    """流式生成RAG回答"""
    # 1. 检索阶段（无法流式，但可以先发送状态）
    yield f"data: {json.dumps({'type': 'status', 'content': '正在检索相关文档...'})}\n\n"
    contexts = await retrieve(question)
    yield f"data: {json.dumps({'type': 'status', 'content': f'找到 {len(contexts)} 条相关文档'})}\n\n"

    # 2. 流式生成阶段
    yield f"data: {json.dumps({'type': 'start', 'content': ''})}\n\n"
    stream = await client.chat.completions.create(
        model="gpt-4o",
        messages=[
            {"role": "system", "content": f"根据以下上下文回答问题：\n{contexts}"},
            {"role": "user", "content": question},
        ],
        stream=True,
    )
    async for chunk in stream:
        if chunk.choices[0].delta.content:
            content = chunk.choices[0].delta.content
            yield f"data: {json.dumps({'type': 'token', 'content': content})}\n\n"

    # 3. 完成
    yield f"data: {json.dumps({'type': 'done', 'content': ''})}\n\n"

                    break;
            }
        }

5. 成本优化

LLM调用是RAG系统最大的成本项。以下是经过实践验证的优化策略：

Embedding缓存

import hashlib
import redis
import numpy as np

class EmbeddingCache:
    """缓存Embedding结果，避免重复计算"""

    def __init__(self, redis_client: redis.Redis, ttl: int = 86400 * 7):
        self.redis = redis_client
        self.ttl = ttl

    def get_or_compute(self, texts: list[str], embed_fn) -> list[list[float]]:
        """批量获取embedding，优先从缓存取"""
        results = [None] * len(texts)
        uncached_indices = []
        uncached_texts = []

        # 查询缓存
        for i, text in enumerate(texts):
            key = f"emb:{hashlib.sha256(text.encode()).hexdigest()}"
            cached = self.redis.get(key)
            if cached:
                results[i] = np.frombuffer(cached, dtype=np.float32).tolist()
            else:
                uncached_indices.append(i)
                uncached_texts.append(text)

        # 批量计算未缓存的
        if uncached_texts:
            new_embeddings = embed_fn(uncached_texts)
            for idx, emb in zip(uncached_indices, new_embeddings):
                results[idx] = emb
                key = f"emb:{hashlib.sha256(texts[idx].encode()).hexdigest()}"
                self.redis.setex(key, self.ttl, np.array(emb, dtype=np.float32).tobytes())

        return results  # type: ignore

Prompt压缩

def compress_context(contexts: list[str], max_tokens: int = 2000) -> str:
    """压缩上下文，保留最相关的内容"""
    # 策略1: 按相关性分数截断（需要检索时返回分数）
    # 策略2: 提取关键句子而非全段
    total = ""
    for ctx in contexts:
        if count_tokens(total + ctx) > max_tokens:
            # 截断到最近的句号
            remaining = max_tokens - count_tokens(total)
            truncated = truncate_to_tokens(ctx, remaining, break_at="。")
            total += truncated
            break
        total += ctx + "\n\n"
    return total

def count_tokens(text: str) -> int:
    """粗略估算token数（中文约1.5字/token）"""
    return len(text) // 2

class TwoStageGenerator:
    """两阶段生成：小模型初筛 + 大模型精排"""

    def __init__(self):

        self.large_model = "gpt-4o"        # 贵但准确

        complexity_prompt = f"""判断以下问题的复杂度：
- simple: 事实查询、单跳推理
- complex: 多跳推理、比较分析、需要综合多个信息

        complexity = await call_llm(
            model=self.small_model,
            prompt=complexity_prompt,
            max_tokens=10,

        # 阶段2: 根据复杂度选择模型

                model=self.small_model,
                prompt=build_rag_prompt(question, contexts),
            )

            # 复杂问题用大模型保证质量
            return await call_llm(
                model=self.large_model,
                prompt=build_rag_prompt(question, contexts),
            )

6. 安全与合规

Prompt注入防御

Prompt注入是RAG系统面临的最大安全风险。攻击者可能通过精心构造的文档或查询来操控LLM输出。

import re

class PromptInjectionDefender:
    """Prompt注入防御层"""

    # 输入侧：检测恶意查询
    INJECTION_PATTERNS = [
        r"ignore\s+(all\s+)?(previous|above)\s+instructions",
        r"你现在是",
        r"忽略.*指令",
        r"system\s*prompt",
        r"<\|im_start\|>",
        r"act\s+as\s+(?:a\s+)?(?:DAN|jailbreak)",
    ]

    def scan_input(self, query: str) -> tuple[bool, str]:
        """扫描用户输入是否包含注入尝试"""
        query_lower = query.lower()
        for pattern in self.INJECTION_PATTERNS:
            if re.search(pattern, query_lower, re.IGNORECASE):
                return True, f"检测到潜在注入: {pattern}"
        return False, ""

    def scan_output(self, answer: str, contexts: list[str]) -> tuple[bool, str]:
        """扫描LLM输出是否泄露系统信息"""
        leak_patterns = [
            r"我的系统提示是",
            r"system prompt",
            r"我的指令是",
            r"I was told to",
        ]
        for pattern in leak_patterns:
            if re.search(pattern, answer, re.IGNORECASE):
                return True, "输出可能包含系统信息泄露"
        return False, ""

    def sanitize_context(self, context: str) -> str:
        """清理检索到的文档内容"""
        # 移除可能的注入指令
        context = re.sub(r"\[SYSTEM\].*?(\n|$)", "", context, flags=re.IGNORECASE)
        context = re.sub(r"", "", context, flags=re.DOTALL)
        return context.strip()

# 在RAG pipeline中使用
defender = PromptInjectionDefender()

def safe_rag_query(question: str) -> str:
    # 1. 输入检查
    is_injection, reason = defender.scan_input(question)
    if is_injection:
        return "抱歉，您的提问中包含不安全的内容，请重新提问。"

    # 2. 检索
    contexts = retriever.invoke(question)

    # 3. 清理检索结果
    clean_contexts = [defender.sanitize_context(c) for c in contexts]

    # 4. 生成
    answer = llm.invoke(question, clean_contexts)

    # 5. 输出检查
    has_leak, reason = defender.scan_output(answer, clean_contexts)
    if has_leak:
        answer = "抱歉，系统无法生成合适的回答，请重新提问。"

    return answer

PII脱敏

# pip install presidio-analyzer presidio-anonymizer
from presidio_analyzer import AnalyzerEngine
from presidio_anonymizer import AnonymizerEngine
from presidio_anonymizer.entities import OperatorConfig

class PIIDetector:
    """使用 Microsoft Presidio 进行PII脱敏"""

    def __init__(self):
        self.analyzer = AnalyzerEngine()
        self.anonymizer = AnonymizerEngine()

    def detect_and_anonymize(self, text: str, language: str = "en") -> str:
        """检测并匿名化PII信息"""
        results = self.analyzer.analyze(
            text=text,
            language=language,
            entities=[
                "PERSON", "EMAIL_ADDRESS", "PHONE_NUMBER",
                "CREDIT_CARD", "IP_ADDRESS", "LOCATION",
            ],
        )

        anonymized = self.anonymizer.anonymize(
            text=text,
            analyzer_results=results,
            operators={
                "PERSON": OperatorConfig("replace", {"new_value": "[姓名]"}),
                "EMAIL_ADDRESS": OperatorConfig("replace", {"new_value": "[邮箱]"}),
                "PHONE_NUMBER": OperatorConfig("replace", {"new_value": "[电话]"}),
                "CREDIT_CARD": OperatorConfig("replace", {"new_value": "[银行卡]"}),
            },
        )
        return anonymized.text

Embedding模型升级（Re-index策略）

更换Embedding模型意味着所有文档需要重新向量化。这是一次高风险操作，需要谨慎规划。

class EmbeddingMigration:
    """Embedding模型迁移管理"""

    def __init__(self, old_model: str, new_model: str, vector_store):
        self.old_model = old_model
        self.new_model = new_model
        self.vector_store = vector_store

    async def reindex(self, documents: list[dict], batch_size: int = 100):
        """重新索引所有文档"""
        new_collection = f"docs_{self.new_model.replace('/', '_')}"

        print(f"开始re-index到集合: {new_collection}")
        total = len(documents)

        for i in range(0, total, batch_size):
            batch = documents[i:i + batch_size]
            # 使用新模型生成embedding
            embeddings = await embed_batch(batch, model=self.new_model)
            # 写入新集合
            await self.vector_store.upsert(
                collection=new_collection,
                ids=[d["id"] for d in batch],
                vectors=embeddings,
                payloads=batch,
            )
            print(f"  进度: {min(i + batch_size, total)}/{total}")

        print(f"Re-index完成，共 {total} 条文档")

    async def validate(self, test_queries: list[dict], new_collection: str):
        """验证新索引质量"""
        hits = 0
        for tq in test_queries:
            results = await self.vector_store.search(
                collection=new_collection,
                query=tq["query"],
                top_k=5,
            )
            if any(r["id"] in tq["expected_ids"] for r in results):
                hits += 1
        hit_rate = hits / len(test_queries)
        print(f"验证 Hit Rate@5: {hit_rate:.2%}")
        return hit_rate

# 迁移流程
# 1. 用新模型re-index所有文档到新集合
# 2. 验证新索引质量
# 3. 灰度切换：先将10%流量导向新索引
# 4. 对比评估指标，确认无退化
# 5. 全量切换，保留旧索引一周后删除

Prompt版本管理

class PromptVersionManager:
    """Prompt版本管理器"""

    def __init__(self):
        self.prompts: dict[str, dict] = {}
        self.active_version: dict[str, str] = {}  # prompt_name -> version

    def register(self, name: str, version: str, template: str, metadata: dict = None):
        """注册新版本的Prompt"""
        key = f"{name}:{version}"
        self.prompts[key] = {
            "template": template,
            "metadata": metadata or {},
            "created_at": datetime.utcnow().isoformat(),
        }
        print(f"注册 Prompt: {key}")

    def activate(self, name: str, version: str):
        """激活指定版本"""
        key = f"{name}:{version}"
        if key not in self.prompts:
            raise ValueError(f"Prompt {key} 不存在")
        self.active_version[name] = version
        print(f"激活 Prompt: {key}")

    def get(self, name: str, version: str = None) -> str:
        """获取Prompt模板"""
        ver = version or self.active_version.get(name)
        if not ver:
            raise ValueError(f"Prompt {name} 未指定版本且无激活版本")
        return self.prompts[f"{name}:{ver}"]["template"]

    def rollback(self, name: str):
        """回滚到上一个版本"""
        current = self.active_version.get(name)
        if not current:
            raise ValueError(f"Prompt {name} 无激活版本")
        # 找到当前版本之前的版本
        versions = sorted(
            [k.split(":")[1] for k in self.prompts if k.startswith(f"{name}:")],
            reverse=True,
        )
        for ver in versions:
            if ver < current:
                self.activate(name, ver)
                return
        raise ValueError(f"没有更早的版本可以回滚")

回滚机制

class RollbackManager:
    """RAG系统回滚管理"""

    def __init__(self):
        self.versions: dict[str, dict] = {}  # version -> config

    def snapshot(self, version: str, config: dict):
        """保存当前系统配置快照"""
        self.versions[version] = {
            "config": config,
            "timestamp": datetime.utcnow().isoformat(),
            "status": "inactive",
        }

    def activate(self, version: str):
        """激活指定版本"""
        for v in self.versions:
            self.versions[v]["status"] = "inactive"
        self.versions[version]["status"] = "active"

    def rollback_to_last_stable(self):
        """回滚到最近的稳定版本"""
        stable_versions = [
            v for v, cfg in self.versions.items()
            if cfg.get("metadata", {}).get("stable", False)
        ]
        if not stable_versions:
            raise RuntimeError("没有可用的稳定版本")
        latest_stable = sorted(stable_versions)[-1]
        self.activate(latest_stable)
        print(f"已回滚到: {latest_stable}")
        return latest_stable

  • Prompt回滚要同步更新评估基线
  • 建议每次变更都创建带时间戳的配置快照
