经典模型架构详解

从 Transformer 原始论文到 LLaMA 系列，手写每个架构的 PyTorch 教学代码

1. Transformer 原始架构 Vaswani 2017

Transformer 是所有现代 LLM 的基石，其核心创新是用 Self-Attention 完全替代 RNN/CNN，实现全局依赖建模。

核心组件

Multi-Head Self-Attention：将 Q/K/V 分成多个头并行计算注意力，拼接后线性投影。
Position-wise FFN：两层全连接网络，中间用 ReLU 激活，维度先扩大 4 倍再缩回。
Add & Norm：残差连接 + Layer Normalization。

Multi-Head Self-Attention 实现

import torch
import torch.nn as nn
import math

class MultiHeadAttention(nn.Module):
    def __init__(self, d_model, n_heads):
        super().__init__()
        self.n_heads = n_heads
        self.d_k = d_model // n_heads
        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, d_model)
        self.W_v = nn.Linear(d_model, d_model)
        self.W_o = nn.Linear(d_model, d_model)

    def forward(self, x, mask=None):
        B, L, D = x.shape
        # 线性投影并拆分为多头: (B, L, D) -> (B, H, L, d_k)
        Q = self.W_q(x).view(B, L, self.n_heads, self.d_k).transpose(1, 2)
        K = self.W_k(x).view(B, L, self.n_heads, self.d_k).transpose(1, 2)
        V = self.W_v(x).view(B, L, self.n_heads, self.d_k).transpose(1, 2)
        # 缩放点积注意力
        scores = (Q @ K.transpose(-2, -1)) / math.sqrt(self.d_k)
        if mask is not None:
            scores = scores.masked_fill(mask == 0, -1e9)
        attn = torch.softmax(scores, dim=-1)
        out = (attn @ V).transpose(1, 2).contiguous().view(B, L, D)
        return self.W_o(out)

Transformer Encoder Layer

class TransformerEncoderLayer(nn.Module):
    def __init__(self, d_model=512, n_heads=8, d_ff=2048, dropout=0.1):
        super().__init__()
        self.attn = MultiHeadAttention(d_model, n_heads)
        self.norm1 = nn.LayerNorm(d_model)
        self.norm2 = nn.LayerNorm(d_model)
        self.ffn = nn.Sequential(
            nn.Linear(d_model, d_ff), nn.ReLU(), nn.Dropout(dropout),
            nn.Linear(d_ff, d_model), nn.Dropout(dropout),
        )

    def forward(self, x, mask=None):
        # Self-Attention + Residual + LayerNorm
        attn_out = self.attn(x, mask)
        x = self.norm1(x + attn_out)
        # FFN + Residual + LayerNorm
        ffn_out = self.ffn(x)
        x = self.norm2(x + ffn_out)
        return x
原始 Transformer 使用 Post-LN（先残差再归一化），GPT 系列使用 Pre-LN（先归一化再残差），后者训练更稳定。

2. BERT Encoder-Only

BERT（Bidirectional Encoder Representations from Transformers）是 Encoder-Only 架构的代表，通过 MLM（Masked Language Model）和 NSP（Next Sentence Prediction）进行双向预训练。

特性说明
架构Transformer Encoder（双向注意力）
预训练任务MLM（随机遮盖 15% 的 token）+ NSP
位置编码可学习的绝对位置编码
典型应用文本分类、NER、问答、句子相似度

BERT 模型简化实现

class BERTEmbedding(nn.Module):
    def __init__(self, vocab_size, d_model=768, max_len=512):
        super().__init__()
        self.tok_emb = nn.Embedding(vocab_size, d_model)
        self.pos_emb = nn.Embedding(max_len, d_model)  # 可学习位置编码
        self.seg_emb = nn.Embedding(2, d_model)       # 句子 A/B 段编码
        self.norm = nn.LayerNorm(d_model)

    def forward(self, input_ids, segment_ids):
        L = input_ids.size(1)
        pos = torch.arange(L, device=input_ids.device).unsqueeze(0)
        return self.norm(self.tok_emb(input_ids)
                       + self.pos_emb(pos)
                       + self.seg_emb(segment_ids))

class BertModel(nn.Module):
    def __init__(self, vocab_size=30522, d_model=768,
                 n_heads=12, n_layers=12, d_ff=3072):
        super().__init__()
        self.embedding = BERTEmbedding(vocab_size, d_model)
        self.layers = nn.ModuleList([
            TransformerEncoderLayer(d_model, n_heads, d_ff)
            for _ in range(n_layers)
        ])
        # MLM 头：预测被遮盖的 token
        self.mlm_head = nn.Sequential(
            nn.Linear(d_model, d_model), nn.GELU(),
            nn.LayerNorm(d_model), nn.Linear(d_model, vocab_size),
        )

    def forward(self, input_ids, segment_ids, mask=None):
        x = self.embedding(input_ids, segment_ids)
        for layer in self.layers:
            x = layer(x, mask)
        logits = self.mlm_head(x)
        return logits  # (B, L, vocab_size)
BERT 的 MLM 任务会随机替换 15% 的 token：其中 80% 替换为 [MASK]，10% 替换为随机 token，10% 保持不变，以缓解预训练与微调之间的分布差异。

3. GPT-2 Decoder-Only

GPT 系列采用 Causal（因果）注意力掩码，每个 token 只能看到自己及之前的 token，通过自回归方式逐 token 生成文本。

Decoder-Only 核心差异

1. 使用 Causal Mask（下三角矩阵）确保自回归特性
2. Pre-LN 结构（LayerNorm 在 Attention/FFN 之前）
3. 训练目标：给定前缀预测下一个 token（Next Token Prediction）

GPT-2 Block 实现

class CausalSelfAttention(nn.Module):
    def __init__(self, d_model, n_heads, max_len=1024):
        super().__init__()
        self.n_heads = n_heads
        self.d_k = d_model // n_heads
        self.c_attn = nn.Linear(d_model, 3 * d_model)  # Q/K/V 合并
        self.c_proj = nn.Linear(d_model, d_model)
        # 因果掩码：下三角矩阵
        self.register_buffer("mask",
            torch.tril(torch.ones(max_len, max_len)).view(1, 1, max_len, max_len))

    def forward(self, x):
        B, L, D = x.shape
        qkv = self.c_attn(x).chunk(3, dim=-1)
        Q, K, V = [t.view(B, L, self.n_heads, self.d_k).transpose(1, 2) for t in qkv]
        scores = (Q @ K.transpose(-2, -1)) / math.sqrt(self.d_k)
        scores = scores.masked_fill(self.mask[:, :, :L, :L] == 0, -1e9)
        attn = torch.softmax(scores, dim=-1)
        out = (attn @ V).transpose(1, 2).contiguous().view(B, L, D)
        return self.c_proj(out)

class GPT2Block(nn.Module):
    def __init__(self, d_model=768, n_heads=12, d_ff=3072):
        super().__init__()
        self.ln1 = nn.LayerNorm(d_model)   # Pre-LN
        self.attn = CausalSelfAttention(d_model, n_heads)
        self.ln2 = nn.LayerNorm(d_model)
        self.mlp = nn.Sequential(
            nn.Linear(d_model, d_ff), nn.GELU(),  # GPT-2 用 GELU
            nn.Linear(d_ff, d_model),
        )

    def forward(self, x):
        x = x + self.attn(self.ln1(x))  # Pre-LN + Residual
        x = x + self.mlp(self.ln2(x))
        return x
GPT-2 的 token embedding 和最终 lm_head 权重是共享的（weight tying），这可以大幅减少参数量。BERT 的 MLM 头则通常不共享权重。

4. T5 Encoder-Decoder

T5（Text-to-Text Transfer Transformer）将所有 NLP 任务统一为 text-to-text 格式：输入是文本，输出也是文本。例如分类任务输出 "positive" 或 "negative" 字符串。

任务T5 输入格式T5 输出格式
翻译"translate English to German: Hello""Hallo"
摘要"summarize: <长文本>""<摘要>"
分类"sst2 sentence: This is great""positive"

T5 Block（含 Encoder-Decoder Attention）

class T5Block(nn.Module):
    def __init__(self, d_model=768, n_heads=12, is_decoder=False):
        super().__init__()
        self.is_decoder = is_decoder
        self.self_attn = MultiHeadAttention(d_model, n_heads)
        self.norm1 = nn.LayerNorm(d_model)
        if is_decoder:
            # Decoder 额外的 Cross-Attention（关注 Encoder 输出）
            self.cross_attn = MultiHeadAttention(d_model, n_heads)
            self.norm_cross = nn.LayerNorm(d_model)
        self.ffn = nn.Sequential(
            nn.Linear(d_model, d_model * 4), nn.ReLU(),
            nn.Linear(d_model * 4, d_model),
        )
        self.norm2 = nn.LayerNorm(d_model)

    def forward(self, x, enc_out=None, self_mask=None):
        # 自注意力（Decoder 使用因果掩码）
        x = x + self.self_attn(self.norm1(x), self_mask)
        if self.is_decoder and enc_out is not None:
            # 交叉注意力：Q 来自 Decoder，K/V 来自 Encoder
            x = x + self.cross_attn(
                self.norm_cross(x), enc_out # 需要修改 MHA 支持 kv 输入
            )
        x = x + self.ffn(self.norm2(x))
        return x
T5 使用 相对位置编码（Relative Position Bias），在注意力分数中加入可学习的偏置项，泛化到更长序列的能力优于绝对位置编码。

5. LLaMA 系列 现代 Decoder-Only

LLaMA（Large Language Model Meta AI）系列引入了多项关键改进：RMSNorm、SwiGLU 激活和 RoPE 旋转位置编码，成为当前最主流的架构范式。

组件LLaMA 创新相比 Transformer 原版
归一化RMSNorm（Pre-Norm）LayerNorm（Post-Norm）
激活函数SwiGLUReLU / GELU
位置编码RoPE（旋转位置编码）绝对/相对位置编码
注意力GQA（Llama 2/3）MHA

RMSNorm 实现

class RMSNorm(nn.Module):
    """Root Mean Square Layer Normalization（无需均值中心化）"""
    def __init__(self, d_model, eps=1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(d_model))
        self.eps = eps

    def forward(self, x):
        # RMS = sqrt(mean(x^2))，比 LayerNorm 少一次均值计算
        rms = torch.sqrt(x.pow(2).mean(-1, keepdim=True) + self.eps)
        return (x / rms) * self.weight

SwiGLU 激活函数

class SwiGLU(nn.Module):
    """SwiGLU: Swish(xW1) * xW3，门控线性单元"""
    def __init__(self, d_model, d_ff):
        super().__init__()
        self.w1 = nn.Linear(d_model, d_ff, bias=False)  # Gate
        self.w3 = nn.Linear(d_model, d_ff, bias=False)  # Up
        self.w2 = nn.Linear(d_ff, d_model, bias=False)  # Down

    def forward(self, x):
        # Swish = x * sigmoid(x)，即 F.silu
        return self.w2(F.silu(self.w1(x)) * self.w3(x))

RoPE 旋转位置编码（简化实现）

def apply_rope(q, k, freqs_cis):
    """将位置信息通过复数旋转注入 Q 和 K"""
    # 将最后一维视为复数对: (re, im)
    q_complex = torch.view_as_complex(q.float().view(*q.shape[:-1], -1, 2))
    k_complex = torch.view_as_complex(k.float().view(*k.shape[:-1], -1, 2))
    # 复数乘法 = 旋转
    q_out = torch.view_as_real(q_complex * freqs_cis).flatten(-2)
    k_out = torch.view_as_real(k_complex * freqs_cis).flatten(-2)
    return q_out.type_as(q), k_out.type_as(k)

def precompute_freqs_cis(dim, max_len, theta=10000.0):
    # 频率: theta_i = 1 / (10000^(2i/d))
    freqs = 1.0 / (theta ** (torch.arange(0, dim, 2).float() / dim))
    t = torch.arange(max_len)
    freqs = torch.outer(t, freqs)  # (max_len, dim//2)
    return torch.polar(torch.ones_like(freqs), freqs)  # e^(i*theta)

完整 LLaMA Block

import torch.nn.functional as F

class LLaMABlock(nn.Module):
    def __init__(self, d_model=4096, n_heads=32, d_ff=11008):
        super().__init__()
        self.attn_norm = RMSNorm(d_model)
        self.attn = CausalSelfAttention(d_model, n_heads)
        self.ffn_norm = RMSNorm(d_model)
        self.ffn = SwiGLU(d_model, d_ff)

    def forward(self, x, freqs_cis=None):
        # Pre-RMSNorm + Causal Attention + Residual
        h = x + self.attn(self.attn_norm(x))
        # Pre-RMSNorm + SwiGLU FFN + Residual
        out = h + self.ffn(self.ffn_norm(h))
        return out
SwiGLU 的 d_ff 通常设为 (2/3) * 4 * d_model（而不是原版的 4 倍），因为有 3 个线性层，保持总参数量相当。LLaMA 7B 的 d_ff=11008 就是这个比例。

从 Transformer 到 LLaMA 的演进趋势：Post-LN -> Pre-LN -> RMSNorm；ReLU -> GELU -> SwiGLU；绝对位置编码 -> RoPE。这些改进共同提升了训练稳定性和模型性能。

快速搭建一个 LLaMA 风格模型

class MiniLLaMA(nn.Module):
    """教学用小型 LLaMA 模型"""
    def __init__(self, vocab_size=32000, d_model=512,
                 n_heads=8, n_layers=6, d_ff=1365, max_len=2048):
        super().__init__()
        self.tok_emb = nn.Embedding(vocab_size, d_model)
        self.layers = nn.ModuleList([
            LLaMABlock(d_model, n_heads, d_ff) for _ in range(n_layers)
        ])
        self.norm = RMSNorm(d_model)
        self.lm_head = nn.Linear(d_model, vocab_size, bias=False)
        # 权重共享
        self.lm_head.weight = self.tok_emb.weight

    def forward(self, input_ids):
        x = self.tok_emb(input_ids)
        for layer in self.layers:
            x = layer(x)
        x = self.norm(x)
        return self.lm_head(x)

# 使用示例
model = MiniLLaMA(vocab_size=32000, d_model=512, n_layers=6)
x = torch.randint(0, 32000, (2, 128))  # batch=2, seq_len=128
logits = model(x)  # (2, 128, 32000)
print(f"参数量: {sum(p.numel() for p in model.parameters()) / 1e6:.1f}M")
